% --- Uniform-scale helper for TikZ figures ---------------------------------
% Renders the content scaled by a single shared factor (\figscale) relative to
% its own natural width, so that text appears at a consistent size across all
% diagrams (instead of \resizebox{\textwidth} which couples text size to the
% diagram's native width). The widest diagram should reach ~\textwidth.
\newsavebox{\figmeasbox}
\newcommand{\figscale}{0.75}
\newcommand{\figuni}[1]{%
  \sbox{\figmeasbox}{#1}%
  \ifdim\dimexpr\figscale\wd\figmeasbox\relax>\textwidth
    \resizebox{\textwidth}{!}{\usebox{\figmeasbox}}%
  \else
    \resizebox{\dimexpr\figscale\wd\figmeasbox\relax}{!}{\usebox{\figmeasbox}}%
  \fi
}
